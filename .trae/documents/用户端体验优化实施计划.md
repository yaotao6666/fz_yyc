### 阶段1：微信授权登录完善

#### 1.1 创建授权登录工具
**文件**：`miniprogram/src/utils/useAuth.ts`

**功能**：
- 自动调用微信登录获取code
- 首次登录自动注册用户
- 二次进入自动获取token
- openid作为唯一标识

**代码实现**：
```typescript
import { ref } from 'vue'

const token = ref(uni.getStorageSync('token') || '')
const userInfo = ref(uni.getStorageSync('userInfo') || null)
const isLoggedIn = ref(!!token.value)
const openid = ref(uni.getStorageSync('openid') || '')

interface LoginResponse {
  token: string
  user: {
    id: number
    openid: string
    nickname: string
  }
}

export function useAuth() {
  const login = async (): Promise<boolean> => {
    try {
      // 已登录则直接返回
      if (token.value && openid.value) {
        return true
      }

      // 获取微信授权码
      const { code } = await uni.login({ provider: 'weixin' })

      if (!code) {
        console.error('获取微信授权码失败')
        return false
      }

      // 调用后端微信登录接口
      const res = await uni.request({
        url: 'http://localhost:8080/api/v1/auth/user/wechat-login',
        method: 'POST',
        data: { code },
        header: {
          'Content-Type': 'application/json'
        }
      }) as { data: LoginResponse }

      if (res.data?.token) {
        token.value = res.data.token
        userInfo.value = res.data.user
        openid.value = res.data.user.openid
        isLoggedIn.value = true

        uni.setStorageSync('token', res.data.token)
        uni.setStorageSync('userInfo', res.data.user)
        uni.setStorageSync('openid', res.data.user.openid)

        console.log('微信登录成功', res.data.token)
        return true
      }

      return false
    } catch (error) {
      console.error('登录失败:', error)
      return false
    }
  }

  const logout = () => {
    token.value = ''
    userInfo.value = null
    openid.value = ''
    isLoggedIn.value = false

    uni.removeStorageSync('token')
    uni.removeStorageSync('userInfo')
    uni.removeStorageSync('openid')
  }

  const ensureAuth = async (): Promise<boolean> => {
    if (!isLoggedIn.value) {
      return await login()
    }
    return true
  }

  return {
    token,
    userInfo,
    openid,
    isLoggedIn,
    login,
    logout,
    ensureAuth
  }
}
```

func WechatLogin(c *gin.Context) {
	var req WechatLoginRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.Fail(c, http.StatusBadRequest, response.CodeParamError, "参数错误")
		return
	}

	// 使用code生成openid（实际项目中应调用微信API）
	openID := "wx_" + req.Code

	// 查找或创建用户
	var user models.User
	result := database.DB.Where("openid = ?", openID).First(&user)

	if result.Error == gorm.ErrRecordNotFound {
		// 首次登录，创建用户
		user = models.User{
			OpenID:   openID,
			Nickname: "微信用户",
			Status:   1,
		}
		database.DB.Create(&user)
	} else if result.Error != nil {
		response.Fail(c, http.StatusInternalServerError, response.CodeServerError, "登录失败")
		return
	}

	// 生成JWT token
	token, _ := utils.GenerateToken(user.ID, "user", user.Nickname)

	response.Success(c, gin.H{
		"token": token,
		"user": gin.H{
			"id":       user.ID,
			"openid":   user.OpenID,
			"nickname": user.Nickname,
		},
	})
}

### 阶段4：配送距离下拉优化

#### 4.1 修改确认订单页面
**文件**：`miniprogram/src/pages/store/confirm.vue`

**功能**：
- 获取商家配送规则
- 生成距离下拉选项
- 添加超出距离风险提示

**代码修改**：
```typescript
import { getStoreDeliveryRules } from '@api'

const deliveryRules = ref<{ distance: number; label: string }[]>([])
const deliveryDistanceIndex = ref(0)
const maxSafeDistance = ref(3) // 建议配送距离

async function loadDeliveryRules() {
  try {
    const rules = await getStoreDeliveryRules(merchantId.value)

    const maxDistance = rules.max_distance || 5
    maxSafeDistance.value = Math.floor(maxDistance * 0.8 * 10) / 10

    // 生成距离选项（0.5km递增）
    for (let d = 0.5; d <= maxDistance; d += 0.5) {
      const isWarning = d > maxSafeDistance.value
      deliveryRules.value.push({
        distance: d,
        label: `${d.toFixed(1)}km${isWarning ? ' ⚠️' : ''}`
      })
    }
  } catch (error) {
    console.error('获取配送规则失败:', error)
    // 使用默认选项
    for (let d = 0.5; d <= 5; d += 0.5) {
      deliveryRules.value.push({
        distance: d,
        label: `${d.toFixed(1)}km`
      })
    }
  }
}

function onDistanceChange(e: any) {
  deliveryDistanceIndex.value = e.detail.value
  deliveryDistance.value = deliveryRules.value[e.detail.value].distance
}

// 模板修改
<view class="form-item">
  <view class="form-label">配送距离</view>
  <picker
    mode="selector"
    :range="deliveryRules"
    range-key="label"
    :value="deliveryDistanceIndex"
    @change="onDistanceChange"
  >
    <view class="picker-value">
      {{ deliveryRules[deliveryDistanceIndex]?.label || '请选择距离' }}
    </view>
  </picker>
  <view class="distance-tip" v-if="deliveryDistance > maxSafeDistance">
    ⚠️ 超出建议配送距离，可能无法配送
  </view>
</view>
```

**代码实现**（伪代码）：
```go
func CreatePayment(c *gin.Context) {
	orderID := c.Param("order_id")

	var order models.Order
	if err := database.DB.First(&order, orderID).Error; err != nil {
		response.Fail(c, http.StatusNotFound, response.CodeOrderNotFound, "订单不存在")
		return
	}

	// 获取商家信息
	var merchant models.Merchant
	database.DB.First(&merchant, order.MerchantID)

	// 调用服务商微信支付
	paymentParams := callServiceProviderPayment(merchant.SubMchID, order.OrderNo, order.PayAmount)

	response.Success(c, paymentParams)
}
```

    // 调用下单接口

```markdown
## 2.2.10 用户授权登录流程

### 功能描述
用户进入商家店铺时，自动完成微信授权登录，获取openid作为唯一身份标识。

### 登录流程
1. 用户进入店铺首页
2. 前端自动调用 `uni.login()` 获取微信授权码
3. 调用后端微信登录接口，传入授权码
4. 后端根据授权码获取openid
5. 查询或创建用户记录
6. 返回JWT token和用户信息
7. 前端保存token，后续请求携带token

### openid管理
- openid是用户在微信生态下的唯一标识
- 不同小程序的openid不同，但同一小程序的openid不变
- 用户二次访问时，直接通过openid识别用户，无需重新授权

### 隐私保护
- 仅获取用户openid，不获取手机号等敏感信息
- 用户昵称默认显示为"微信用户"
- 后续可扩展用户主动完善个人信息

## 3.2.18 微信支付（服务商模式）

**接口路径**：`POST /api/v1/user/orders/{order_id}/pay`

**功能**：
- 创建微信支付订单
- 使用服务商模式，支持多商家收款
- 返回调起支付所需参数

**请求参数**：无

**响应**：
```json
{
  "code": 0,
  "data": {
    "order_id": "order_123",
    "pay_params": {
      "appId": "wx_appid",
      "timeStamp": "1234567890",
      "nonceStr": "random_string",
      "package": "prepay_id=xxx",
      "signType": "MD5",
      "paySign": "signature"
    }
  }
}
```

**服务商模式说明**：
- 服务商拥有微信支付商户号
- 每个商家是服务商的子商户（sub_mch_id）
- 支付资金直接结算到商家的子商户账户
- 服务商收取手续费，商家收到税后金额
```
