   ```go
     productID := c.Param("product_id")

     if err := database.DB.Where("id = ? AND merchant_id = ?", id, merchantID).Preload("Category").Preload("Specs").First(&product).Error; err != nil {
   ```

#### 问题定位
经过分析，接口调用链路完整且正确。404错误可能由以下原因导致：

1. **商品不存在**：数据库中确实没有该商品
2. **权限验证**：商品不属于当前登录的商家（`merchant_id` 不匹配）
3. **JWT Token问题**：请求未携带有效的认证Token
4. **商品ID错误**：传递的商品ID不正确

#### 验证步骤
需要检查：
1. 点击商品图片时传递的ID是否正确
2. JWT Token是否正确携带
3. 商品是否属于当前商家

```go
func GetProduct(c *gin.Context) {
	merchantID := middleware.GetMerchantID(c)
	productID := c.Param("product_id")
	id, _ := strconv.ParseUint(productID, 10, 64)

	var product models.Product
	if err := database.DB.Where("id = ?", id).First(&product).Error; err != nil {
		if err == gorm.ErrRecordNotFound {
			response.Fail(c, http.StatusNotFound, response.CodeNotFound, "商品不存在")
		} else {
			response.Fail(c, http.StatusInternalServerError, response.CodeServerError, "查询商品失败")
		}
		return
	}

	// 验证商品归属
	if product.MerchantID != merchantID {
		response.Fail(c, http.StatusForbidden, response.CodeForbidden, "无权访问该商品")
		return
	}

	// 加载关联数据
	database.DB.Preload("Category").Preload("Specs").First(&product, id)

	response.Success(c, product)
}
```
